# Balloon texts: key = text. The keys are the "# @balloon" marks in tapes/.
newline = Ctrl-J: new line, ↑: line up
history_ml = ↑: the whole statement again
edit_any = Edit any line, Enter runs it
editor = Multi-line editor
highlight = Syntax highlighting
ctrlc = Ctrl-C clears the statement
tab = Tab completion
alias = Columns after an alias
hint = Gray hint: → accepts it
history = Statement history: ↑
search = Ctrl-R: search the history
plain = Plain isql output
preview = SET PREVIEW: compact table
csv = IMPORT INTO from CSV
parquet = Parquet and Avro too
prepare = PREPARE + EXEC SQL
execfrom = EXEC SQL ... FROM a file
colors_builtin = Built-in colors
colors_own = Your colors: HQISQL_COLORS
